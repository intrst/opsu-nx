cmake_minimum_required(VERSION 3.14)
project(opsu_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB_RECURSE OPSU_CORE_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(opsu_core ${OPSU_CORE_SRC})
target_include_directories(opsu_core PUBLIC include)
find_package(Threads REQUIRED)
target_link_libraries(opsu_core PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(opsu_core PRIVATE /W4)
    # /fp:precise is MSVC default; keep it explicit so float results stay identical to Java
    target_compile_options(opsu_core PUBLIC /fp:precise)
else()
    target_compile_options(opsu_core PRIVATE -Wall -Wextra)
    # keep float results identical to Java: GCC contracts a*b+c into FMA by default on ARM
    # (and on x86 with -march=native), which changes the last bits of slider curves.
    target_compile_options(opsu_core PUBLIC -ffp-contract=off)
endif()
